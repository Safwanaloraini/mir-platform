   394→function StatusChangeButton({ task, canApprove, canEdit, onChanged }: { task: any; canApprove: boolean; canEdit: boolean; onChanged: () => void }) {
   395→  const { toast } = useToast();
   396→  const [open, setOpen] = useState(false);
   397→  const [status, setStatus] = useState("");
   398→  const [note, setNote] = useState("");
   399→  const [stallReason, setStallReason] = useState("");
   400→  const [submitting, setSubmitting] = useState(false);
   401→
   402→  const transitions = STATUS_TRANSITIONS[task.status] ?? [];
   403→  const allowed = transitions.filter((s) => s !== "completed_approved" || canApprove);
   404→
   405→  if (allowed.length === 0 || !canEdit) return null;
   406→
   407→  const submit = async () => {
   408→    if (!status) return;
   409→    if (status === "stalled" && !stallReason.trim()) {
   410→      toast({ title: "سبب التعثر مطلوب", variant: "destructive" });

   700→function ChecklistCard({ task, canEdit, onChanged }: { task: any; canEdit: boolean; onChanged: () => void }) {
   701→  const { toast } = useToast();
   702→  const [newText, setNewText] = useState("");
   703→  const [newRequired, setNewRequired] = useState(false);
   704→
   705→  const items = task.checklist ?? [];
   706→  const doneCount = items.filter((i: any) => i.done).length;
   707→  const requiredTotal = items.filter((i: any) => i.required).length;
   708→  const requiredDone = items.filter((i: any) => i.required && i.done).length;
   709→
   710→  const toggle = async (item: any) => {
   711→    try {
   712→      await apiFetch(`/api/tasks/${task.id}/checklist`, {
   713→        method: "POST",
   714→        body: JSON.stringify({ id: item.id, done: !item.done }),
   715→      });
   716→      onChanged();
   717→    } catch (e: any) {
   718→      toast({ title: "تعذّر التحديث", description: e?.message, variant: "destructive" });
   719→    }
   720→  };
   721→
   722→  const add = async () => {
   723→    if (!newText.trim()) return;
   724→    try {
   725→      await apiFetch(`/api/tasks/${task.id}/checklist`, {
   726→        method: "POST",
   727→        body: JSON.stringify({ text: newText, required: newRequired }),
   728→      });
   729→      setNewText(""); setNewRequired(false);
   730→      onChanged();
   731→    } catch (e: any) {
   732→      toast({ title: "تعذّر الإضافة", description: e?.message, variant: "destructive" });
   733→    }
   734→  };
   735→
   736→  return (
   737→    <SectionCard
   738→      title={`قائمة التحقق (${doneCount}/${items.length})`}
   739→      action={requiredTotal > 0 ? <Badge variant="outline" className="text-xs">إلزامي: {requiredDone}/{requiredTotal}</Badge> : undefined}
   740→    >
   741→      {items.length > 0 ? (
   742→        <div className="space-y-1.5 mb-3">
   743→          {items.map((item: any) => (
   744→            <div key={item.id} className="flex items-center gap-2 p-2 rounded hover:bg-accent group">
   745→              <Checkbox checked={item.done} onCheckedChange={() => toggle(item)} />
   746→              <span className={`text-sm flex-1 ${item.done ? "line-through text-muted-foreground" : ""}`}>{item.text}</span>
   747→              {item.required && (
   748→                <Badge variant="outline" className={`text-[10px] ${item.done ? "text-green-700 border-green-300" : "text-red-700 border-red-300"}`}>
   749→                  {item.done ? "منجز" : "إلزامي"}
   750→                </Badge>
   751→              )}
   752→            </div>
   753→          ))}
   754→        </div>
   755→      ) : (
   756→        <div className="text-xs text-muted-foreground mb-3">لا توجد عناصر تحقق بعد</div>
   757→      )}
   758→      {canEdit && (
   759→        <div className="flex items-center gap-2 pt-2 border-t border-border">
   760→          <Input
   761→            placeholder="أضف عنصر تحقق…"
   762→            value={newText}
   763→            onChange={(e) => setNewText(e.target.value)}

   858→    setSubmitting(true);
   859→    setUploadProgress(`جارٍ رفع: ${file.name}…`);
   860→
   861→    try {
   862→      // 1) ارفع الملف إلى /api/files/upload
   863→      const formData = new FormData();
   864→      formData.append("file", file);
   865→      const uploadRes = await fetch("/api/files/upload", { method: "POST", body: formData });
   866→      const uploadData = await uploadRes.json();
   867→      if (!uploadRes.ok) throw new Error(uploadData?.error || "فشل الرفع");
   868→
   869→      // 2) أنشئ سجل المرفق مرتبطًا بالملف المرفوع
   870→      await apiFetch(`/api/tasks/${taskId}/attachments`, {
   871→        method: "POST",
   872→        body: JSON.stringify({
   873→          fileName: uploadData.fileName,
   874→          fileUrl: uploadData.url,
   875→          fileType: uploadData.fileType,
   876→          fileSize: uploadData.fileSize,
   877→          required,
   878→        }),
   879→      });
   880→
   881→      toast({ title: "تم رفع المرفق بنجاح", description: uploadData.fileName });
   882→      setOpen(false);
   883→      setRequired(false);
   884→      onAdded();
   885→    } catch (err: any) {
   886→      toast({ title: "تعذّر رفع الملف", description: err?.message, variant: "destructive" });
   887→    } finally {
   888→      setSubmitting(false);
   889→      setUploadProgress("");
   890→      if (fileInputRef.current) fileInputRef.current.value = "";
   891→    }
   892→  };
   893→
   894→  return (
   895→    <>
   896→      <Button variant="ghost" size="sm" onClick={() => setOpen(true)} className="text-xs gap-1">
   897→        <Plus className="h-3.5 w-3.5" /> إضافة
   898→      </Button>
   899→      <Dialog open={open} onOpenChange={(v) => { setOpen(v); if (!v) { setRequired(false); setUploadProgress(""); } }}>
   900→        <DialogContent>
   901→          <DialogHeader>
   902→            <DialogTitle>رفع مرفق من الجهاز</DialogTitle>
   903→            <DialogDescription>اختر ملفًا من جهازك (صور PNG/JPEG/GIF/WebP، PDF، Word، Excel) — الحد الأقصى 100 ميجابايت</DialogDescription>
   904→          </DialogHeader>
   905→          <div className="space-y-4">
   906→            {/* منطقة اختيار الملف */}
   907→            <div
   908→              onClick={() => !submitting && fileInputRef.current?.click()}
   909→              className="border-2 border-dashed border-border rounded-xl p-8 text-center cursor-pointer hover:border-primary hover:bg-accent/50 transition"
   910→            >
   911→              <Upload className="h-10 w-10 mx-auto text-muted-foreground mb-3" />
   912→              {submitting ? (
   913→                <div className="space-y-2">
   914→                  <Loader2 className="h-5 w-5 mx-auto animate-spin text-primary" />
   915→                  <p className="text-sm text-muted-foreground">{uploadProgress}</p>
   916→                </div>
   917→              ) : (
   918→                <>
   919→                  <p className="text-sm font-medium">اضغط لاختيار ملف</p>
   920→                  <p className="text-xs text-muted-foreground mt-1">أو اسحب الملف هنا</p>
   921→                  <p className="text-[11px] text-muted-foreground/70 mt-2">
   922→                    الأنواع المدعومة: جميع الصور (PNG, JPEG, GIF, WebP, SVG)، PDF، Word، Excel، PowerPoint، نصوص، ZIP
   923→                  </p>
   924→                </>
   925→              )}
   926→              <input
   927→                ref={fileInputRef}
   928→                type="file"
   929→                className="hidden"
   930→                accept="image/*,application/pdf,.doc,.docx,.xls,.xlsx,.ppt,.pptx,.txt,.csv,.zip,.rar,.rtf,.odt,.ods,.odp"
   931→                onChange={handleFileChange}
   932→                disabled={submitting}
   933→              />
   934→            </div>
   935→
   936→            <label className="flex items-center gap-2 text-sm cursor-pointer">
   937→              <Checkbox checked={required} onCheckedChange={(v) => setRequired(!!v)} />
   938→              مرفق إلزامي (مطلوب لاعتماد الإكمال)
   939→            </label>
   940→          </div>
   941→          <DialogFooter>
   942→            <DialogClose asChild><Button variant="ghost" disabled={submitting}>إلغاء</Button></DialogClose>
   943→          </DialogFooter>
   944→        </DialogContent>
   945→      </Dialog>
   946→    </>
   947→  );
